import { createRoot } from 'react-dom/client';
import type { Root } from 'react-dom/client';

export type ModalMode = 'setup' | 'unlock';

interface ModalControls {
    show: (mode: ModalMode, errorHint?: string) => Promise<string | null>;
    /** Immediately resolve any pending prompt with null (used on unmount). */
    cancel: () => void;
}

// ─── component ───────────────────────────────────────────────────────────────

interface Props {
    onRegister: (controls: ModalControls) => void;
}

const PasswordModalComponent: React.FC<Props> = ({ onRegister }) => {
    const [visible, setVisible] = useState(false);
    const [mode, setMode] = useState<ModalMode>('unlock');
    const [password, setPassword] = useState('');
    const [confirm, setConfirm] = useState('');
    const [error, setError] = useState('');
    const resolveRef = useRef<((val: string | null) => void) | null>(null);
    const inputRef = useRef<HTMLInputElement>(null);

    useEffect(() => {
        onRegister({
            show: (m, errorHint) =>
                new Promise<string | null>((resolve) => {
                    resolveRef.current = resolve;
                    setPassword('');
                    setConfirm('');
                    setError(errorHint ?? '');
                    setMode(m);
                    setVisible(true);
                }),

            // BUG-13: resolve any pending promise with null so callers don't hang on unmount
            cancel: () => {
                if (resolveRef.current) {
                    resolveRef.current(null);
                    resolveRef.current = null;
                }
                setVisible(false);
            },
        });
    }, [onRegister]);

    useEffect(() => {
        if (visible) setTimeout(() => inputRef.current?.focus(), 50);
    }, [visible]);

    const dismiss = useCallback((value: string | null) => {
        resolveRef.current?.(value);
        resolveRef.current = null;
        setVisible(false);
    }, []);

    const handleSubmit = useCallback(() => {
        if (mode === 'setup') {
            if (password.length < 8) {
                setError('Password must be at least 8 characters');
                return;
            }
            if (password !== confirm) {
                setError('Passwords do not match');
                return;
            }
        } else {
            if (!password) {
                setError('Please enter your password');
                return;
            }
        }
        dismiss(password);
    }, [mode, password, confirm, dismiss]);

    const handleKeyDown = useCallback(
        (e: React.KeyboardEvent) => {
            if (e.key === 'Enter') handleSubmit();
            if (e.key === 'Escape') dismiss(null);
        },
        [handleSubmit, dismiss],
    );

    if (!visible) return null;

    const isSetup = mode === 'setup';

    return (
        <div style={S.overlay}>
            <div style={S.dialog} onKeyDown={handleKeyDown}>
                {/* Header */}
                <div style={{ marginBottom: '24px' }}>
                    <div style={{ fontSize: '20px', fontWeight: 700, marginBottom: '8px' }}>
                        {isSetup ? 'Set Up Encryption' : 'Vault is Locked'}
                    </div>
                    <div style={{ fontSize: '14px', color: 'var(--nh-text-secondary, #a6adc8)', lineHeight: 1.5 }}>
                        {isSetup
                            ? 'Create a master password. All .md files will be encrypted transparently with AES-256-GCM.'
                            : 'Enter your master password to decrypt vault notes.'}
                    </div>
                </div>

                {/* Password field */}
                <div style={{ marginBottom: '14px' }}>
                    <label style={S.label}>{isSetup ? 'New Password' : 'Password'}</label>
                    <input
                        ref={inputRef}
                        type="password"
                        value={password}
                        onChange={(e) => { setPassword(e.target.value); setError(''); }}
                        placeholder={isSetup ? 'At least 8 characters' : 'Master password'}
                        style={S.input}
                        autoComplete={isSetup ? 'new-password' : 'current-password'}
                    />
                </div>

                {/* Confirm field (setup only) */}
                {isSetup && (
                    <div style={{ marginBottom: '14px' }}>
                        <label style={S.label}>Confirm Password</label>
                        <input
                            type="password"
                            value={confirm}
                            onChange={(e) => { setConfirm(e.target.value); setError(''); }}
                            placeholder="Repeat password"
                            style={S.input}
                            autoComplete="new-password"
                        />
                    </div>
                )}

                {/* Error message */}
                {error && <div style={S.error}>{error}</div>}

                {/* Actions */}
                <div style={{ display: 'flex', gap: '10px', justifyContent: 'flex-end', marginTop: '6px' }}>
                    <button style={S.btnSecondary} onClick={() => dismiss(null)}>
                        {isSetup ? 'Skip for Now' : 'Work Offline'}
                    </button>
                    <button style={S.btnPrimary} onClick={handleSubmit}>
                        {isSetup ? 'Enable Encryption' : 'Unlock Vault'}
                    </button>
                </div>
            </div>
        </div>
    );
};

// ─── singleton manager ────────────────────────────────────────────────────────

let root: Root | null = null;
let container: HTMLDivElement | null = null;
let controls: ModalControls | null = null;
let pendingResolvers: Array<(c: ModalControls) => void> = [];

// BUG-9: Serialise concurrent prompt() calls so only one modal is active at a time.
// If a second caller arrives while the first modal is open, it waits for the first
// to resolve before showing its own modal.
let activePromise: Promise<string | null> | null = null;

export const PasswordModal = {
    mount() {
        if (container) return;
        container = document.createElement('div');
        container.id = 'vault-lock-modal-root';
        document.body.appendChild(container);
        root = createRoot(container);
        root.render(
            <PasswordModalComponent
                onRegister={(c) => {
                    controls = c;
                    pendingResolvers.forEach((resolve) => resolve(c));
                    pendingResolvers = [];
                }}
            />,
        );
    },

    unmount() {
        // BUG-13: cancel any in-flight prompt so its Promise resolves (null) rather
        // than hanging forever after the component is torn down.
        controls?.cancel();
        root?.unmount();
        root = null;
        container?.remove();
        container = null;
        controls = null;
        pendingResolvers = [];
        activePromise = null;
    },

    /** Show the modal and resolve with the entered password, or null if cancelled. */
    async prompt(mode: ModalMode, errorHint?: string): Promise<string | null> {
        // BUG-9: wait for any currently active prompt to finish before showing a new one
        while (activePromise) {
            await activePromise.catch(() => {});
        }

        let c = controls;
        if (!c) {
            c = await new Promise<ModalControls>((resolve) => pendingResolvers.push(resolve));
        }

        const p = c.show(mode, errorHint);
        activePromise = p;
        try {
            return await p;
        } finally {
            activePromise = null;
        }
    },
};
